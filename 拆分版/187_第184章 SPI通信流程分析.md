```c
int spi_sync(struct spi_device *spi, struct spi_message *message)
{
    int ret;
    //锁定SPI 控制器的总线锁互斥体
    mutex_lock(&spi->controller->bus_lock_mutex);
    //执行同步SPI 传输
    ret = __spi_sync(spi, message);
    //解锁SPI 控制器的总线锁互斥体
    mutex_unlock(&spi->controller->bus_lock_mutex);
    return ret;
}
```
