```c
static int rockchip_canfd_probe(struct platform_device *pdev) {
        // 定义用于存储网络设备、rockchip_canfd 结构体、资源、内存地址以及错误码和中断号的变量
        struct net_device *ndev;
        struct rockchip_canfd *rcan;
        struct resource *res;
        void __iomem *addr;
        int err, irq;
        // 从平台设备获取中断号
        irq = platform_get_irq(pdev, 0);
        // 若获取中断号失败
        if (irq < 0) {
            // 向设备日志输出错误信息
            dev_err(&pdev->dev, "未能获取有效的中断号\n");
            // 返回设备不存在的错误码
            return -ENODEV;
        }
        // 获取平台设备的内存资源
        res = platform_get_resource(pdev, IORESOURCE_MEM, 0);
        // 将获取的内存资源映射到内核虚拟地址空间
        addr = devm_ioremap_resource(&pdev->dev, res);
        // 若映射操作失败
        if (IS_ERR(addr))
            // 返回设备忙的错误码
        return -EBUSY;
```

在代码清单228.5 中最重要的是使用register_candev()函数注册CAN 设备，函数实现如代码清单228.6 所示：代码清单228.6
```c
int register_candev(struct net_device *dev) {
        // 获取网络设备dev 的私有数据，即指向can_priv 结构体的指针priv
    struct can_priv *priv = netdev_priv(dev);
        /* 确保termination_const、termination_const_cnt 和do_set_termination 的一致性。
         * 这三个成员要么都被设置，要么都未被设置。
         */
        // 如果termination_const 和termination_const_cnt 的设置状态不一致，
        // 或者termination_const 和do_set_termination 的设置状态不一致
        if ((!priv->termination_const !=!priv->termination_const_cnt) ||
            (!priv->termination_const !=!priv->do_set_termination))
            // 返回无效参数错误码，表示参数设置不符合要求
            return -EINVAL;
        // 如果bitrate_const 和bitrate_const_cnt 的设置状态不一致
        if (!priv->bitrate_const !=!priv->bitrate_const_cnt)
            // 返回无效参数错误码
            return -EINVAL;
        // 如果data_bitrate_const 和data_bitrate_const_cnt 的设置状态不一致
        if (!priv->data_bitrate_const !=!priv->data_bitrate_const_cnt)
            // 返回无效参数错误码
            return -EINVAL;
        // 设置网络设备dev 的RTNL（Routing Netlink）链路操作集为can_link_ops
        // can_link_ops 定义了与CAN 设备链路相关的操作
        dev->rtnl_link_ops = &can_link_ops;
        // 关闭网络设备dev 的载波信号，通常表示设备尚未连接或未准备好传输数据
    netif_carrier_off(dev);
        // 调用register_netdev 函数注册网络设备dev
        // 如果注册成功，返回0；否则返回相应的错误码
        return register_netdev(dev);
}
// 将register_candev 函数导出，供其他内核模块在遵循GPL 协议的前提下使用
EXPORT_SYMBOL_GPL(register_candev);
```

```c
// 定义一个函数register_netdev，用于注册网络设备，参数为指向网络设备结构体的指针dev
int register_netdev(struct net_device *dev) {
        int err;
        // 以可被信号中断的方式获取RTNL（Routing Netlink）锁
        // 如果获取锁失败（例如被信号中断），返回-EINTR 错误码
        if (rtnl_lock_killable())
            return -EINTR;
        // 调用另一个名为register_netdevice()函数
        // 将返回值存储在err 变量中
        err = register_netdevice(dev);
        // 释放之前获取的RTNL 锁
        rtnl_unlock();
        // 返回注册操作的结果（成功返回0，失败返回相应错误码）
        return err;
}
// 将register_netdev 函数导出，供其他内核模块使用
EXPORT_SYMBOL(register_netdev);
```
